name: Create QA to PROD PRs

on:
  workflow_dispatch:
    inputs:
      pr_title:
        description: "Enter PR title"
        required: true
        type: string

permissions:
  contents: read

env:
  GITHUB_ORG: cvs-health-source-code
  SOURCE_BRANCH: release/qa
  TARGET_BRANCH: main
  REPO_FILE: .github/prod-promotion/repos.txt


jobs:

  # ============================================================
  # Job 1: Read repository list and generate matrix
  # ============================================================
  setup-matrix:
    name: Load repository list
    runs-on: self-hosted

    outputs:
      matrix: ${{ steps.read.outputs.matrix }}

    steps:

      - name: Checkout mics-gitops
        uses: actions/checkout@v4
        with:
          ref: main

      - name: Parse repository list
        id: read
        shell: bash
        run: |
          set -euo pipefail

          if [[ ! -f "${REPO_FILE}" ]]; then
            echo "Repository file not found: ${REPO_FILE}"
            exit 1
          fi

          MATRIX=$(python3 <<'PYEOF'
          import json
          import os

          repo_file = os.environ["REPO_FILE"]

          repos = []

          with open(repo_file, "r") as f:
              for line in f:
                  line = line.strip()

                  if not line:
                      continue

                  if line.startswith("#"):
                      continue

                  repos.append({
                      "repo": line
                  })

          if not repos:
              raise SystemExit("No repositories found in repos.txt")

          print(json.dumps({"include": repos}))
          PYEOF
          )

          echo "matrix=$MATRIX" >> "$GITHUB_OUTPUT"

          echo "Repositories selected:"
          echo "$MATRIX" | python3 -m json.tool


  # ============================================================
  # Job 2: Create PR for each repository
  # ============================================================
  create-pr:
    name: PR ${{ matrix.repo }} release/qa to main

    needs:
      - setup-matrix

    runs-on: self-hosted

    strategy:
      fail-fast: false
      matrix: ${{ fromJson(needs.setup-matrix.outputs.matrix) }}

    steps:

      # --------------------------------------------------------
      # Generate GitHub App token
      # --------------------------------------------------------
      - name: Generate GitHub App token
        id: app-token
        uses: actions/create-github-app-token@v1
        with:
          app-id: ${{ secrets.GHA_WORKFLOW_APP_ID }}
          private-key: ${{ secrets.GHA_WORKFLOW_APP_PRIVATE_KEY }}
          owner: cvs-health-source-code


      # --------------------------------------------------------
      # Validate source and target branches
      # --------------------------------------------------------
      - name: Validate branches
        id: validate
        shell: bash
        env:
          GH_TOKEN: ${{ steps.app-token.outputs.token }}
        run: |
          set -euo pipefail

          REPO="${GITHUB_ORG}/${{ matrix.repo }}"

          echo "Repository : $REPO"
          echo "Source     : ${SOURCE_BRANCH}"
          echo "Target     : ${TARGET_BRANCH}"

          if ! gh api \
            "repos/${REPO}/branches/${SOURCE_BRANCH}" \
            >/dev/null 2>&1; then

            echo "Source branch '${SOURCE_BRANCH}' does not exist."
            echo "eligible=false" >> "$GITHUB_OUTPUT"
            exit 0
          fi

          if ! gh api \
            "repos/${REPO}/branches/${TARGET_BRANCH}" \
            >/dev/null 2>&1; then

            echo "Target branch '${TARGET_BRANCH}' does not exist."
            echo "eligible=false" >> "$GITHUB_OUTPUT"
            exit 0
          fi

          echo "eligible=true" >> "$GITHUB_OUTPUT"


      # --------------------------------------------------------
      # Compare release/qa with main
      # --------------------------------------------------------
      - name: Check whether PR is needed
        id: compare
        if: steps.validate.outputs.eligible == 'true'
        shell: bash
        env:
          GH_TOKEN: ${{ steps.app-token.outputs.token }}
        run: |
          set -euo pipefail

          REPO="${GITHUB_ORG}/${{ matrix.repo }}"

          AHEAD=$(gh api \
            "repos/${REPO}/compare/${TARGET_BRANCH}...${SOURCE_BRANCH}" \
            --jq '.ahead_by')

          echo "${SOURCE_BRANCH} is ahead of ${TARGET_BRANCH} by $AHEAD commit(s)."

          if [[ "$AHEAD" -gt 0 ]]; then
            echo "needed=true" >> "$GITHUB_OUTPUT"
          else
            echo "needed=false" >> "$GITHUB_OUTPUT"
          fi


      # --------------------------------------------------------
      # Check whether an open PR already exists
      # --------------------------------------------------------
      - name: Check for existing PR
        id: existing
        if: steps.compare.outputs.needed == 'true'
        shell: bash
        env:
          GH_TOKEN: ${{ steps.app-token.outputs.token }}
        run: |
          set -euo pipefail

          REPO="${GITHUB_ORG}/${{ matrix.repo }}"

          EXISTING=$(gh pr list \
            --repo "$REPO" \
            --base "${TARGET_BRANCH}" \
            --head "${SOURCE_BRANCH}" \
            --state open \
            --json number,url \
            --jq '.[0].number // ""')

          if [[ -n "$EXISTING" ]]; then

            URL=$(gh pr view "$EXISTING" \
              --repo "$REPO" \
              --json url \
              --jq '.url')

            echo "Existing PR found:"
            echo "$URL"

            echo "exists=true" >> "$GITHUB_OUTPUT"
            echo "url=$URL" >> "$GITHUB_OUTPUT"

          else

            echo "No existing PR found."
            echo "exists=false" >> "$GITHUB_OUTPUT"

          fi


      # --------------------------------------------------------
      # Create PR
      # --------------------------------------------------------
      - name: Create PR from release/qa to main
        id: create
        if: |
          steps.compare.outputs.needed == 'true' &&
          steps.existing.outputs.exists != 'true'
        shell: bash
        env:
          GH_TOKEN: ${{ steps.app-token.outputs.token }}
          PR_TITLE: ${{ inputs.pr_title }}
        run: |
          set -euo pipefail

          REPO="${GITHUB_ORG}/${{ matrix.repo }}"

          cat > /tmp/pr-body.md <<'PREOF'
          ## QA to Production Promotion

          This PR promotes the current `release/qa` branch to `main`.

          Source branch: `release/qa`

          Target branch: `main`

          This PR was created automatically by the MICS GitHub Actions workflow.

          No approval or merge has been performed automatically.

          Please review and approve/merge manually after the required validation is completed.
          PREOF

          PR_URL=$(gh pr create \
            --repo "$REPO" \
            --base "${TARGET_BRANCH}" \
            --head "${SOURCE_BRANCH}" \
            --title "$PR_TITLE" \
            --body-file /tmp/pr-body.md)

          echo "Created PR:"
          echo "$PR_URL"

          echo "url=$PR_URL" >> "$GITHUB_OUTPUT"


      # --------------------------------------------------------
      # Determine PR URL to publish
      #
      # This includes:
      #   - newly created PR
      #   - already existing PR
      # --------------------------------------------------------
      - name: Prepare PR link
        id: result
        if: always()
        shell: bash
        run: |
          set -euo pipefail

          URL=""

          if [[ -n "${{ steps.create.outputs.url }}" ]]; then
            URL="${{ steps.create.outputs.url }}"

          elif [[ -n "${{ steps.existing.outputs.url }}" ]]; then
            URL="${{ steps.existing.outputs.url }}"
          fi

          if [[ -n "$URL" ]]; then
            echo "url=$URL" >> "$GITHUB_OUTPUT"

            echo "$URL" > pr-link.txt

            echo "PR available:"
            echo "$URL"
          else
            echo "No PR generated for ${{ matrix.repo }}"
            touch pr-link.txt
          fi


      # --------------------------------------------------------
      # Upload individual PR link
      # Each matrix job gets unique artifact name
      # --------------------------------------------------------
      - name: Upload PR link
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: pr-link-${{ matrix.repo }}
          path: pr-link.txt
          if-no-files-found: ignore
          retention-days: 1


  # ============================================================
  # Job 3: Print all PR URLs on Summary page
  # ============================================================
  summarize-prs:
    name: PR Summary

    needs:
      - create-pr

    if: always()

    runs-on: self-hosted

    steps:

      # --------------------------------------------------------
      # Download all PR-link artifacts
      # --------------------------------------------------------
      - name: Download PR links
        uses: actions/download-artifact@v4
        with:
          pattern: pr-link-*
          path: pr-links
          merge-multiple: true


      # --------------------------------------------------------
      # Print PR links one after another
      # --------------------------------------------------------
      - name: Print PR links in Summary
        shell: bash
        run: |
          set -euo pipefail

          echo "## QA to Production Pull Requests" >> "$GITHUB_STEP_SUMMARY"
          echo "" >> "$GITHUB_STEP_SUMMARY"

          LINKS=$(find pr-links \
            -type f \
            -name "pr-link.txt" \
            -exec cat {} \; 2>/dev/null \
            | sed '/^[[:space:]]*$/d' \
            | sort -u)

          if [[ -z "$LINKS" ]]; then

            echo "No pull requests were created or found." \
              >> "$GITHUB_STEP_SUMMARY"

          else

            while IFS= read -r URL; do

              echo "$URL" >> "$GITHUB_STEP_SUMMARY"
              echo "" >> "$GITHUB_STEP_SUMMARY"

            done <<< "$LINKS"

          fi
